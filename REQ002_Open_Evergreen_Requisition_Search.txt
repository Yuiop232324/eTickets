    WebAutomation.ExecuteJavascript BrowserInstance: iCIMSBrowser Javascript: $'''function ExecuteScript() {
const isVisible = function (element) {
    if (!element) return false;
    const style = element.ownerDocument.defaultView.getComputedStyle(element);
    const box = element.getBoundingClientRect();
    return style.visibility !== \'hidden\' && style.display !== \'none\' && box.width > 0 && box.height > 0;
};
const findIn = function (doc) {
    const byId = doc.getElementById(\'refineSearch_anchor\');
    if (byId && isVisible(byId)) return byId;
    return Array.from(doc.querySelectorAll(\'a, button, [role="button"]\')).find(function (element) {
        const text = (element.innerText || element.textContent || \'\').replace(/\\s+/g, \' \').trim().toLowerCase();
        return isVisible(element) && text === \'search form\';
    });
};
let target = findIn(document);
if (!target) {
    const frames = Array.from(document.querySelectorAll(\'iframe\'));
    for (const frame of frames) {
        try {
            const doc = frame.contentDocument || frame.contentWindow.document;
            target = findIn(doc);
            if (target) break;
        } catch (error) {
        }
    }
}
if (!target) return \'SEARCH_FORM_NOT_FOUND\';
target.scrollIntoView({block: \'center\', inline: \'center\'});
target.click();
return \'SEARCH_FORM_CLICKED\';
}''' Result=> REQ002JsResult
END
